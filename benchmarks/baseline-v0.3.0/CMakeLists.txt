cmake_minimum_required(VERSION 3.25)
project(fixedwide VERSION 0.3.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CheckCXXSourceCompiles)

set(FIXEDWIDE_DECIMALS 12 CACHE STRING "Shared FP64/FP128 fractional digits (one value, 8 through 12)")
set_property(CACHE FIXEDWIDE_DECIMALS PROPERTY STRINGS 8 9 10 11 12)
if(NOT FIXEDWIDE_DECIMALS MATCHES "^(8|9|10|11|12)$")
    message(FATAL_ERROR "FIXEDWIDE_DECIMALS must be one integer from 8 through 12; changing it changes the data contract")
endif()
option(FIXEDWIDE_BUILD_TESTS "Build dependency-free deterministic tests" ${PROJECT_IS_TOP_LEVEL})
option(FIXEDWIDE_BUILD_ORACLE_TESTS "Build Boost.Multiprecision differential tests (test-only dependency)" OFF)
option(FIXEDWIDE_BUILD_BENCHMARKS "Build standalone benchmark harness" OFF)
option(FIXEDWIDE_SANITIZE "Enable AddressSanitizer + UndefinedBehaviorSanitizer" OFF)
option(FIXEDWIDE_COVERAGE "Enable Clang source coverage" OFF)
option(FIXEDWIDE_BUILD_FUZZER "Build Clang libFuzzer target" OFF)
option(FIXEDWIDE_BUILD_EXAMPLES "Build example" ${PROJECT_IS_TOP_LEVEL})

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
check_cxx_source_compiles("#include <expected>\n#ifndef __x86_64__\n#error x86-64 required\n#endif\nint main(){using U=unsigned _BitInt(256); __int128 a=1; std::expected<__int128,int> b=a; U n=4; return int(n / U(*b));}" FIXEDWIDE_COMPILER_CAPABLE)
if(NOT FIXEDWIDE_COMPILER_CAPABLE)
    message(FATAL_ERROR "Requires x86-64, C++23 std::expected, __int128 and C++ _BitInt(256). Use a capable Clang + standard library.")
endif()

configure_file(cmake/config.hpp.in generated/fixedwide/config.hpp @ONLY)
add_library(fixedwide src/arithmetic.cpp src/bigint.cpp src/chars.cpp src/integer_chars.cpp src/floating.cpp src/string.cpp)
add_library(fixedwide::fixedwide ALIAS fixedwide)
target_compile_features(fixedwide PUBLIC cxx_std_23)
target_include_directories(fixedwide PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/generated>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_options(fixedwide PRIVATE -Wall -Wextra -Wconversion -Wshadow)
set_target_properties(fixedwide PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 3 POSITION_INDEPENDENT_CODE ON)
if(FIXEDWIDE_SANITIZE)
    target_compile_options(fixedwide PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
    target_link_options(fixedwide PUBLIC -fsanitize=address,undefined)
endif()
if(FIXEDWIDE_COVERAGE)
    target_compile_options(fixedwide PUBLIC -fprofile-instr-generate -fcoverage-mapping)
    target_link_options(fixedwide PUBLIC -fprofile-instr-generate)
endif()
if(FIXEDWIDE_BUILD_TESTS OR FIXEDWIDE_BUILD_ORACLE_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
if(FIXEDWIDE_BUILD_BENCHMARKS)
    add_subdirectory(benchmarks)
endif()
if(FIXEDWIDE_BUILD_EXAMPLES)
    add_executable(fixedwide_example examples/example.cpp)
    target_link_libraries(fixedwide_example PRIVATE fixedwide::fixedwide)
endif()
if(FIXEDWIDE_BUILD_FUZZER)
    if(NOT FIXEDWIDE_SANITIZE)
        message(FATAL_ERROR "Fuzzing requires FIXEDWIDE_SANITIZE=ON so the library itself is instrumented")
    endif()
    add_executable(fixedwide_fuzz fuzz/fuzz.cpp)
    target_compile_options(fixedwide PRIVATE -fsanitize=fuzzer-no-link)
    target_compile_options(fixedwide_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
    target_link_options(fixedwide_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
    target_link_libraries(fixedwide_fuzz PRIVATE fixedwide::fixedwide)
endif()

install(TARGETS fixedwide EXPORT fixedwideTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/generated/fixedwide/config.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/fixedwide)
install(EXPORT fixedwideTargets NAMESPACE fixedwide:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/fixedwide)
configure_package_config_file(cmake/fixedwideConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/fixedwideConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/fixedwide)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/fixedwideConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/fixedwideConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/fixedwideConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/fixedwide)
